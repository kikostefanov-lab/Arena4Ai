# Phase 0 CLI Orchestrator Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build a working CLI that runs two Claude teams in isolated Docker containers against a YAML brief, streams events to JSONL, cross-judges results, and prints a ranked winner.

**Architecture:** Foundation-first — `@arena/shared` types/schemas/constants first, then `@arena/orchestrator` built bottom-up (state machine → brief parser → Claude adapter → Docker sandbox → judging → competition runner → CLI). Persistence is flat JSONL files in `./runs/<competitionId>/`, no database.

**Tech Stack:** TypeScript 5 strict, npm workspaces, Turborepo 2, Zod 3, Vitest 1, Commander.js 12, js-yaml, dockerode 4, Node.js 20+

```typescript
import { z } from 'zod';
import { CompetitionFormat } from '../constants/formats.js';

const rubricCriterionSchema = z.object({
  id: z.string().min(1),
  description: z.string().min(1),
  weight: z.number().min(0).max(1),
  maxScore: z.number().positive(),
});

const rubricSchema = z.object({
  criteria: z.array(rubricCriterionSchema).min(1),
}).refine(
  (r) => Math.abs(r.criteria.reduce((sum, c) => sum + c.weight, 0) - 1) < 0.001,
  { message: 'Rubric criterion weights must sum to 1' }
);

export const briefSchema = z.object({
  id: z.string().min(1),
  title: z.string().min(1),
  problem: z.string().min(1),
  constraints: z.array(z.string()),
  deliverables: z.array(z.string()).min(1),
  rubric: rubricSchema,
  format: z.nativeEnum(CompetitionFormat),
  timeLimitMs: z.number().positive(),
});

export type BriefInput = z.infer<typeof briefSchema>;
```

**Step 3: Create `packages/orchestrator/src/engine/state-machine.ts`**

```typescript
import { CompetitionState, VALID_TRANSITIONS } from '@arena/shared';

export function transition(
  current: CompetitionState,
  next: CompetitionState
): CompetitionState {
  const allowed = VALID_TRANSITIONS[current];
  if (!allowed.includes(next)) {
    throw new Error(
      `Invalid transition: ${current} → ${next}. Allowed: [${allowed.join(', ')}]`
    );
  }
  return next;
}
```

```typescript
import { EventType } from '@arena/shared';

interface ClockManagerOptions {
  timeLimitMs: number;
  competitionId: string;
}

export class ClockManager extends EventEmitter {
  private timeLimitMs: number;
  private competitionId: string;
  private warningTimer?: ReturnType<typeof setTimeout>;
  private endTimer?: ReturnType<typeof setTimeout>;

  constructor(options: ClockManagerOptions) {
    super();
    this.timeLimitMs = options.timeLimitMs;
    this.competitionId = options.competitionId;
  }

  start(): void {
    const warningMs = this.timeLimitMs * 0.8;

    this.warningTimer = setTimeout(() => {
      this.emit('clockEvent', EventType.TIME_WARNING);
    }, warningMs);

    this.endTimer = setTimeout(() => {
      this.emit('clockEvent', EventType.TIME_UP);
    }, this.timeLimitMs);
  }

  stop(): void {
    clearTimeout(this.warningTimer);
    clearTimeout(this.endTimer);
  }

  getRemainingMs(startedAt: Date): number {
    return Math.max(0, this.timeLimitMs - (Date.now() - startedAt.getTime()));
  }
}
```

```typescript
  const result = briefSchema.safeParse(raw);
```

```typescript
import { EventType } from '@arena/shared';
import type { ArenaEvent } from '@arena/shared';
import { randomUUID } from 'crypto';

interface NormalizerContext {
  competitionId: string;
  teamId: string;
}

export function normalizeLine(
  line: string,
  ctx: NormalizerContext
): ArenaEvent | null {
  if (!line.trim()) return null;

  let parsed: Record<string, unknown>;
  try {
    parsed = JSON.parse(line) as Record<string, unknown>;
  } catch {
    return null;
  }

  const base = {
    eventId: randomUUID(),
    competitionId: ctx.competitionId,
    teamId: ctx.teamId,
    timestamp: new Date().toISOString(),
    metadata: {},
  };

  switch (parsed['type']) {
    case 'tool_use':
      return {
        ...base,
        type: EventType.TOOL_CALL,
        payload: { toolName: parsed['name'], input: parsed['input'] },
      };
    case 'text':
      return {
        ...base,
        type: EventType.REASONING,
        payload: { text: parsed['text'] },
      };
    case 'error':
      return {
        ...base,
        type: EventType.ERROR,
        payload: { message: parsed['message'] ?? 'unknown error' },
      };
    default:
      return null;
  }
}
```

```typescript
import { spawn } from 'child_process';
import type { ChildProcess } from 'child_process';
import { createInterface } from 'readline';
import { BaseAdapter } from '../base-adapter.js';
import { normalizeLine } from './claude-normalizer.js';
import type { Brief, Deliverable } from '@arena/shared';

interface ClaudeAdapterOptions {
  teamId: string;
  competitionId: string;
  workdir: string;              // sandbox working directory
  claudeBin?: string;           // path to claude binary, defaults to 'claude'
}

export class ClaudeAdapter extends BaseAdapter {
  teamId: string;
  private competitionId: string;
  private workdir: string;
  private claudeBin: string;
  private brief?: Brief;
  private persona?: string;
  private process?: ChildProcess;

  constructor(options: ClaudeAdapterOptions) {
    super();
    this.teamId = options.teamId;
    this.competitionId = options.competitionId;
    this.workdir = options.workdir;
    this.claudeBin = options.claudeBin ?? process.env['CLAUDE_BIN'] ?? 'claude';
  }

  async injectBrief(brief: Brief, persona: string): Promise<void> {
    this.brief = brief;
    this.persona = persona;
  }

  async startExecution(): Promise<void> {
    if (!this.brief || !this.persona) {
      throw new Error('Must call injectBrief() before startExecution()');
    }

    const prompt = this.buildPrompt(this.brief, this.persona);

    this.process = spawn(
      this.claudeBin,
      ['--print', '--output-format', 'stream-json', '--no-markdown', '-p', prompt],
      { cwd: this.workdir, stdio: ['ignore', 'pipe', 'pipe'] }
    );

    const rl = createInterface({ input: this.process.stdout! });
    rl.on('line', (line) => {
      const event = normalizeLine(line, {
        competitionId: this.competitionId,
        teamId: this.teamId,
      });
      if (event) this.emitArenaEvent(event);
    });

    return new Promise((resolve, reject) => {
      this.process!.on('close', (code) => {
        if (code === 0 || code === null) resolve();
        else reject(new Error(`Claude process exited with code ${code}`));
      });
      this.process!.on('error', reject);
    });
  }

  async collectDeliverables(): Promise<Deliverable> {
    const { readdir, readFile } = await import('fs/promises');
    const { join } = await import('path');

    const entries = await readdir(this.workdir, { withFileTypes: true });
    const files: Array<{ path: string; content: string }> = [];

    for (const entry of entries) {
      if (entry.isFile()) {
        const content = await readFile(join(this.workdir, entry.name), 'utf-8');
        files.push({ path: entry.name, content });
      }
    }

    return {
      teamId: this.teamId,
      files,
      collectedAt: new Date().toISOString(),
    };
  }

  async shutdown(): Promise<void> {
    if (this.process && !this.process.killed) {
      this.process.kill('SIGTERM');
    }
  }

  private buildPrompt(brief: Brief, persona: string): string {
    return `${persona}

---

# Competition Brief: ${brief.title}

## Problem
${brief.problem}

## Constraints
${brief.constraints.map((c) => `- ${c}`).join('\n')}

## Required Deliverables
${brief.deliverables.map((d) => `- ${d}`).join('\n')}

## Scoring Rubric
${brief.rubric.criteria.map((c) => `- **${c.id}** (weight: ${c.weight}): ${c.description}`).join('\n')}

---

Begin working on the deliverables now. Create all required files in your current directory.`;
  }
}
```

```typescript
import Dockerode from 'dockerode';
import { mkdir, rm } from 'fs/promises';
import { join } from 'path';

const docker = new Dockerode({
  socketPath: process.env['DOCKER_SOCKET'] ?? '/var/run/docker.sock',
});

export interface SandboxConfig {
  teamId: string;
  competitionId: string;
  runsDir: string;
  image?: string;
}

export interface Sandbox {
  teamId: string;
  workdir: string;           // host path mounted into container
  containerId: string;
  stop(): Promise<void>;
  remove(): Promise<void>;
}

export async function createSandbox(config: SandboxConfig): Promise<Sandbox> {
  const image = config.image ?? 'node:20-alpine';
  const workdir = join(config.runsDir, config.competitionId, config.teamId);

  await mkdir(workdir, { recursive: true });

  // Pull image if not present
  await new Promise<void>((resolve, reject) => {
    docker.pull(image, (err: Error | null, stream: NodeJS.ReadableStream) => {
      if (err) return reject(err);
      docker.modem.followProgress(stream, (err: Error | null) => {
        if (err) reject(err);
        else resolve();
      });
    });
  });

  const container = await docker.createContainer({
    Image: image,
    Cmd: ['sleep', 'infinity'],  // keep alive; adapter spawns claude inside
    WorkingDir: '/workspace',
    HostConfig: {
      Binds: [`${workdir}:/workspace`],
      Memory: 512 * 1024 * 1024,    // 512MB
      CpuQuota: 100000,              // 1 CPU
      NetworkMode: 'none',           // no network access
    },
  });

  await container.start();

  return {
    teamId: config.teamId,
    workdir,
    containerId: container.id,

    async stop() {
      await container.stop({ t: 5 }).catch(() => {});
    },

    async remove() {
      await container.remove({ force: true }).catch(() => {});
      await rm(workdir, { recursive: true, force: true });
    },
  };
}
```

```typescript
import type { Deliverable, Rubric, JudgeResult } from '@arena/shared';

export function scoreDeliverable(
  deliverable: Deliverable,
  rubric: Rubric,
  requiredFiles: string[]
): JudgeResult {
  const presentFiles = new Set(deliverable.files.map((f) => f.path));
  const missingCount = requiredFiles.filter((f) => !presentFiles.has(f)).length;
  const completenessScore = requiredFiles.length === 0
    ? 10
    : Math.max(0, 10 - (missingCount / requiredFiles.length) * 10);

  const totalContent = deliverable.files.reduce((sum, f) => sum + f.content.length, 0);
  const qualityScore = Math.min(10, totalContent / 200); // rough proxy: 2000 chars = 10/10

  const scores = rubric.criteria.map((criterion) => {
    let score = 5; // default
    let commentary = 'Automated check — no specific scorer for this criterion.';

    if (criterion.id === 'completeness') {
      score = completenessScore;
      commentary = missingCount === 0
        ? 'All required deliverables present.'
        : `Missing ${missingCount} of ${requiredFiles.length} required files.`;
    } else if (criterion.id === 'quality' || criterion.id === 'correctness') {
      score = qualityScore;
      commentary = `Content volume proxy score (${totalContent} chars).`;
    }

    return { criterionId: criterion.id, score, commentary };
  });

  const overallScore = rubric.criteria.reduce((sum, criterion, i) => {
    return sum + (scores[i]?.score ?? 0) * criterion.weight;
  }, 0);

  return {
    judgeId: 'automated',
    teamId: deliverable.teamId,
    scores,
    overallScore,
  };
}
```

```typescript
import type { JudgeResult, ScoreCard } from '@arena/shared';

export function aggregateScores(teamId: string, judgeResults: JudgeResult[]): ScoreCard {
  if (judgeResults.length === 0) {
    throw new Error(`No judge results for team: ${teamId}`);
  }

  const finalScore =
    judgeResults.reduce((sum, r) => sum + r.overallScore, 0) / judgeResults.length;

  return {
    teamId,
    judgeResults,
    finalScore: Math.round(finalScore * 100) / 100,
    rank: 0, // assigned after all teams scored
  };
}

export function rankScoreCards(scoreCards: ScoreCard[]): ScoreCard[] {
  const sorted = [...scoreCards].sort((a, b) => b.finalScore - a.finalScore);
  return sorted.map((card, i) => ({ ...card, rank: i + 1 }));
}
```

```typescript
import { spawn } from 'child_process';
import type { Deliverable, Rubric, JudgeResult } from '@arena/shared';
import { judgeResultSchema } from '@arena/shared';

export async function aiJudge(
  deliverable: Deliverable,
  rubric: Rubric,
  claudeBin: string = process.env['CLAUDE_BIN'] ?? 'claude'
): Promise<JudgeResult> {
  const prompt = buildJudgePrompt(deliverable, rubric);

  return new Promise((resolve, reject) => {
    const proc = spawn(claudeBin, ['--print', '--no-markdown', '-p', prompt], {
      stdio: ['ignore', 'pipe', 'pipe'],
    });

    let output = '';
    proc.stdout.on('data', (chunk: Buffer) => { output += chunk.toString(); });
    proc.on('close', (code) => {
      if (code !== 0) return reject(new Error(`AI judge process failed: ${code}`));

      // Extract JSON block from output
      const match = output.match(/```json\n?([\s\S]*?)\n?```/) ?? output.match(/(\{[\s\S]*\})/);
      if (!match?.[1]) return reject(new Error('AI judge returned no JSON'));

      try {
        const raw = JSON.parse(match[1]);
        const result = judgeResultSchema.parse({
          ...raw,
          judgeId: `ai-claude`,
          teamId: deliverable.teamId,
        });
        resolve(result as JudgeResult);
      } catch (e) {
        reject(new Error(`Failed to parse judge output: ${e}`));
      }
    });
    proc.on('error', reject);
  });
}

function buildJudgePrompt(deliverable: Deliverable, rubric: Rubric): string {
  const filesSection = deliverable.files
    .map((f) => `### ${f.path}\n\`\`\`\n${f.content}\n\`\`\``)
    .join('\n\n');

  const criteriaList = rubric.criteria
    .map((c) => `- **${c.id}** (max ${c.maxScore} points): ${c.description}`)
    .join('\n');

  return `You are a rigorous technical judge evaluating a competition submission.

## Scoring Rubric
${criteriaList}

## Submission
${filesSection}

## Instructions
Score each criterion from 0 to its maxScore. Be honest and critical.
Return ONLY a JSON object in this exact format:
\`\`\`json
{
  "scores": [
    { "criterionId": "string", "score": number, "commentary": "string" }
  ],
  "overallScore": number
}
\`\`\`
overallScore is the weighted sum of scores (use rubric weights).`;
}
```

```typescript
import { createWriteStream, mkdirSync } from 'fs';
import type { WriteStream } from 'fs';
import { join } from 'path';
import type { ArenaEvent } from '@arena/shared';

export class EventLogger {
  private stream: WriteStream;

  private constructor(stream: WriteStream) {
    this.stream = stream;
  }

  static async create(competitionId: string, runsDir: string): Promise<EventLogger> {
    const dir = join(runsDir, competitionId);
    mkdirSync(dir, { recursive: true });
    const stream = createWriteStream(join(dir, 'events.jsonl'), { flags: 'a' });
    return new EventLogger(stream);
  }

  async log(event: ArenaEvent): Promise<void> {
    return new Promise((resolve, reject) => {
      this.stream.write(JSON.stringify(event) + '\n', (err) => {
        if (err) reject(err);
        else resolve();
      });
    });
  }

  async close(): Promise<void> {
    return new Promise((resolve, reject) => {
      this.stream.end((err?: Error) => {
        if (err) reject(err);
        else resolve();
      });
    });
  }
}
```

```typescript
import { transition } from './state-machine.js';
import { ClockManager } from './clock-manager.js';
import { SandboxManager } from '../sandbox/sandbox-manager.js';
import { ClaudeAdapter } from '../adapters/claude/claude-adapter.js';
import { resolvePersona } from '../adapters/claude/claude-personas.js';
import { EventLogger } from '../events/event-logger.js';
import { scoreDeliverable } from '../judging/rubric-scorer.js';
import { aiJudge } from '../judging/ai-judge.js';
import { aggregateScores, rankScoreCards } from '../judging/score-aggregator.js';
import { printResults } from '../judging/results-reporter.js';
import { writeFile } from 'fs/promises';
import { join } from 'path';
import { randomUUID } from 'crypto';
import {
  CompetitionState,
  EventType,
  type Brief,
  type ScoreCard,
  type ArenaEvent,
} from '@arena/shared';

interface RunOptions {
  brief: Brief;
  teamASpec: string;   // e.g. 'claude:architect'
  teamBSpec: string;   // e.g. 'claude:pragmatist'
  runsDir: string;
  claudeBin?: string;
}

export async function runCompetition(options: RunOptions): Promise<ScoreCard[]> {
  const competitionId = `comp_${Date.now()}_${randomUUID().slice(0, 8)}`;
  let state = CompetitionState.DRAFT;

  console.log(`\n[Arena] Starting competition: ${competitionId}`);
  console.log(`[Arena] Brief: ${options.brief.title}`);

  const logger = await EventLogger.create(competitionId, options.runsDir);

  const logEvent = async (teamId: string, type: EventType, payload: unknown) => {
    await logger.log({
      eventId: randomUUID(),
      competitionId,
      teamId,
      timestamp: new Date().toISOString(),
      type,
      payload,
      metadata: {},
    });
  };

  // DRAFT → CONFIGURED
  state = transition(state, CompetitionState.CONFIGURED);
  console.log(`[Arena] State: ${state}`);

  const teamAPersona = resolvePersona(options.teamASpec);
  const teamBPersona = resolvePersona(options.teamBSpec);

  const sandboxManager = new SandboxManager(competitionId, options.runsDir);

  // CONFIGURED → LAUNCHING
  state = transition(state, CompetitionState.LAUNCHING);
  console.log(`[Arena] State: ${state} — creating Docker sandboxes...`);

  const [sandboxA, sandboxB] = await Promise.all([
    sandboxManager.create('team-a'),
    sandboxManager.create('team-b'),
  ]);

  const adapterA = new ClaudeAdapter({
    teamId: 'team-a',
    competitionId,
    workdir: sandboxA.workdir,
    claudeBin: options.claudeBin,
  });

  const adapterB = new ClaudeAdapter({
    teamId: 'team-b',
    competitionId,
    workdir: sandboxB.workdir,
    claudeBin: options.claudeBin,
  });

  await adapterA.injectBrief(options.brief, teamAPersona.persona);
  await adapterB.injectBrief(options.brief, teamBPersona.persona);

  adapterA.on('arenaEvent', (e: ArenaEvent) => { void logger.log(e); });
  adapterB.on('arenaEvent', (e: ArenaEvent) => { void logger.log(e); });

  // LAUNCHING → RUNNING
  state = transition(state, CompetitionState.RUNNING);
  console.log(`[Arena] State: ${state} — both teams running (${options.brief.timeLimitMs / 60000} min)...`);
  await logEvent('system', EventType.COMPETITION_START, { competitionId });

  const clock = new ClockManager({
    timeLimitMs: options.brief.timeLimitMs,
    competitionId,
  });

  await new Promise<void>((resolve) => {
    clock.on('clockEvent', async (type: EventType) => {
      await logEvent('system', type, {});
      if (type === EventType.TIME_UP) {
        console.log('[Arena] Time is up!');
        resolve();
      } else if (type === EventType.TIME_WARNING) {
        console.log('[Arena] 80% time elapsed — warning sent.');
      }
    });
    clock.start();

    // Run both teams in parallel; resolve on TIME_UP (clock above) or both complete
    Promise.all([
      adapterA.startExecution().catch((e: Error) => console.error('[team-a]', e.message)),
      adapterB.startExecution().catch((e: Error) => console.error('[team-b]', e.message)),
    ]).then(() => { clock.stop(); resolve(); });
  });

  await adapterA.shutdown();
  await adapterB.shutdown();

  // TIME_UP → COLLECTING
  state = transition(state, CompetitionState.TIME_UP);
  state = transition(state, CompetitionState.COLLECTING);
  console.log(`[Arena] State: ${state} — collecting deliverables...`);

  const [deliverableA, deliverableB] = await Promise.all([
    adapterA.collectDeliverables(),
    adapterB.collectDeliverables(),
  ]);

  // COLLECTING → JUDGING
  state = transition(state, CompetitionState.JUDGING);
  console.log(`[Arena] State: ${state} — judging...`);

  const requiredFiles = options.brief.deliverables;

  const [autoScoreA, autoScoreB] = [
    scoreDeliverable(deliverableA, options.brief.rubric, requiredFiles),
    scoreDeliverable(deliverableB, options.brief.rubric, requiredFiles),
  ];

  console.log('[Arena] Running AI cross-judge...');
  const [aiScoreA, aiScoreB] = await Promise.all([
    aiJudge(deliverableA, options.brief.rubric, options.claudeBin).catch((e: Error) => {
      console.warn('[Arena] AI judge failed for team-a:', e.message);
      return autoScoreA; // fallback to automated score
    }),
    aiJudge(deliverableB, options.brief.rubric, options.claudeBin).catch((e: Error) => {
      console.warn('[Arena] AI judge failed for team-b:', e.message);
      return autoScoreB;
    }),
  ]);

  const scoreCardA = aggregateScores('team-a', [autoScoreA, aiScoreA]);
  const scoreCardB = aggregateScores('team-b', [autoScoreB, aiScoreB]);
  const rankedCards = rankScoreCards([scoreCardA, scoreCardB]);

  // JUDGING → SCORED → COMPLETE
  state = transition(state, CompetitionState.SCORED);
  state = transition(state, CompetitionState.COMPLETE);

  await logEvent('system', EventType.COMPETITION_COMPLETE, { rankedCards });
  await logger.close();

  // Save results
  await writeFile(
    join(options.runsDir, competitionId, 'results.json'),
    JSON.stringify(rankedCards, null, 2)
  );

  await sandboxManager.removeAll();

  printResults(rankedCards);

  return rankedCards;
}
```
